cmake_minimum_required(VERSION 3.16)
project(obuina VERSION 0.2.0 LANGUAGES C)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  set(BT_TOP_LEVEL ON)
else()
  set(BT_TOP_LEVEL OFF)
endif()
option(BT_BUILD_TESTS "Build numerical regression tests" ${BT_TOP_LEVEL})
option(BT_SANITISE "Enable address and undefined behaviour sanitisers" OFF)
set(BT_CURUPIXA_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../curupixa" CACHE PATH "Curupixa source directory")
if(NOT EXISTS "${BT_CURUPIXA_SOURCE_DIR}/lib/continuous_traits.h" OR
   NOT EXISTS "${BT_CURUPIXA_SOURCE_DIR}/lib/reconstruction.h")
  message(FATAL_ERROR "Obuina requires curupixa's trait reconstruction/prepared likelihood APIs (a1adfc4 or newer)")
endif()
if(NOT EXISTS "${BT_CURUPIXA_SOURCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Set BT_CURUPIXA_SOURCE_DIR to a curupixa source checkout")
endif()
set(BUILD_SHARED_LIBS OFF)
set(CURUPIXA_BUILD_TESTS OFF)
set(CURUPIXA_INSTALL OFF)
add_subdirectory("${BT_CURUPIXA_SOURCE_DIR}" curupixa)
add_library(obuina SHARED src/version.c src/trees.c src/continuous.c src/discrete.c src/batch.c)
add_library(obuina::obuina ALIAS obuina)
target_compile_features(obuina PUBLIC c_std_17)
set_target_properties(obuina PROPERTIES C_EXTENSIONS OFF VERSION ${PROJECT_VERSION} SOVERSION 0)
target_include_directories(obuina PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<BUILD_INTERFACE:${BT_CURUPIXA_SOURCE_DIR}/lib>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/curupixa>)
target_link_libraries(obuina PRIVATE curupixa::curupixa m)
add_executable(obuina_cli src/cli.c)
set_target_properties(obuina_cli PROPERTIES OUTPUT_NAME obuina C_EXTENSIONS OFF)
if(UNIX AND NOT APPLE AND NOT IS_ABSOLUTE "${CMAKE_INSTALL_LIBDIR}")
  set_target_properties(obuina_cli PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
endif()
target_link_libraries(obuina_cli PRIVATE obuina m)
add_executable(obuina_example examples/basic.c)
target_link_libraries(obuina_example PRIVATE obuina m)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  foreach(target obuina obuina_cli obuina_example)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endforeach()
  if(BT_SANITISE)
    foreach(target curupixa obuina obuina_cli obuina_example)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endforeach()
  endif()
endif()
if(BT_BUILD_TESTS)
  enable_testing()
  foreach(name trees continuous discrete)
    add_executable(test_${name} tests/test_${name}.c)
    target_link_libraries(test_${name} PRIVATE obuina m)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
      target_compile_options(test_${name} PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
      if(BT_SANITISE)
        target_compile_options(test_${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(test_${name} PRIVATE -fsanitize=address,undefined)
      endif()
    endif()
    add_test(NAME ${name} COMMAND test_${name})
  endforeach()
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND AND NOT BT_SANITISE)
    add_test(NAME interfaces COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_interfaces.py
      $<TARGET_FILE:obuina_cli> $<TARGET_FILE:obuina> ${CMAKE_CURRENT_SOURCE_DIR})
    add_test(NAME missing_interfaces COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_missing_interfaces.py
      $<TARGET_FILE:obuina_cli> $<TARGET_FILE:obuina> ${CMAKE_CURRENT_SOURCE_DIR})
  endif()
  add_test(NAME c_example COMMAND obuina_example)
endif()
install(TARGETS obuina obuina_cli EXPORT obuinaTargets
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/obuina.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES "${BT_CURUPIXA_SOURCE_DIR}/lib/tree.h" "${BT_CURUPIXA_SOURCE_DIR}/lib/status.h"
  DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/curupixa)
install(FILES python/obuina.py DESTINATION ${CMAKE_INSTALL_DATADIR}/obuina/python)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/obuina)
install(EXPORT obuinaTargets NAMESPACE obuina:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/obuina)
configure_package_config_file(cmake/obuinaConfig.cmake.in
  "${CMAKE_CURRENT_BINARY_DIR}/obuinaConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/obuina)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/obuinaConfigVersion.cmake"
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/obuinaConfig.cmake"
  "${CMAKE_CURRENT_BINARY_DIR}/obuinaConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/obuina)
